## 通过 npm 集成依赖

在项目的 `package.json` 文件中添加以下依赖项：

```json
{
  "devDependencies": {
    "electron": "24.8.8"
  },
  "dependencies": {
    "fcr-ui-scene": "^3.1.0",
    "agora-edu-core": "^3.1.0",
    "agora-rte-sdk": "^3.1.0",
    "agora-foundation": "^1.0.0",
    "agora-ui-foundation": "^1.0.0",
    "@electron/remote": "2.0.1",
    "agora-electron-sdk": "4.3.2-build.90-rc.1",
    "electron-screenshots": "^0.5.26",
    "jszip": "^3.10.1"
  }
}
```

然后执行：

```shell
npm install
```
