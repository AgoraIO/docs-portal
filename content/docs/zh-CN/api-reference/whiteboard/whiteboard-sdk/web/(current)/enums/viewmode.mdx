export declare enum ViewMode {
    /**
     * 自由模式。不跟随任何人，也不被任何人跟随，自己可以自由操作视角。
     */
    Freedom = "freedom",
    /**
     * 跟随模式。跟随当前房间的主播视角。
     */
    Follower = "follower",
    /**
     * 主播模式。可以自由操作视角，但房间中处于跟随模式的人会跟随。
     */
    Broadcaster = "broadcaster",
}
