factory ChatTranslateLanguage.fromJson(Map map) {
  String code = map["code"];
  String name = map["name"];
  String nativeName = map["nativeName"];
  return ChatTranslateLanguage._private(
    languageCode: code,
    languageName: name,
    languageNativeName: nativeName,
  );
}
